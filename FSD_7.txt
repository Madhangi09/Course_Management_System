// Mock Fetch API
const mockFetch = (url) => {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      try {
        let data;

        if (url === "/api/students") {
          data = mockDatabase.students;
        } else if (url === "/api/courses") {
          data = mockDatabase.courses;
        } else if (url === "/api/enrollments") {
          data = mockDatabase.enrollments;
        } else {
          throw new Error("API endpoint not found");
        }

        resolve({
          ok: true,
          json: async () => data,
        });
      } catch (error) {
        reject(error);
      }
    }, 800);
  });
};

/* =========================
   CONTEXT API
   ========================= */

const AppContext = createContext();

const AppProvider = ({ children }) => {
  const [student, setStudent] = useState(null);
  const [courses, setCourses] = useState([]);
  const [enrollments, setEnrollments] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [isLoggedIn, setIsLoggedIn] = useState(true);

  const fetchData = async () => {
    try {
      setLoading(true);
      setError("");

      const [studentsResponse, coursesResponse, enrollmentResponse] =
        await Promise.all([
          mockFetch("/api/students"),
          mockFetch("/api/courses"),
          mockFetch("/api/enrollments"),
        ]);

      if (
        !studentsResponse.ok ||
        !coursesResponse.ok ||
        !enrollmentResponse.ok
      ) {
        throw new Error("Failed to fetch API data");
      }

      const students = await studentsResponse.json();
      const coursesData = await coursesResponse.json();
      const enrollmentData = await enrollmentResponse.json();

      setStudent(students[0]);
      setCourses(coursesData);
      setEnrollments(enrollmentData);
    } catch (err) {
      setError("Unable to load data from mock API.");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, []);

  const logout = () => {
    setIsLoggedIn(false);
  };

  const login = () => {
    setIsLoggedIn(true);
    fetchData();
  };

  return (
    <AppContext.Provider
      value={{
        student,
        courses,
        enrollments,
        loading,
        error,
        isLoggedIn,
        login,
        logout,
        refreshData: fetchData,
      }}
    >
      {children}
    </AppContext.Provider>
  );
};

/* =========================
   DASHBOARD
   ========================= */

const Dashboard = () => {
  const {
    student,
    courses,
    enrollments,
    loading,
    error,
    logout,
    refreshData,
  } = useContext(AppContext);

  if (loading) {
    return (
      <div className="center">
        <div className="loader"></div>
        <h2>Loading data...</h2>
        <p>Fetching information from Mock API</p>
      </div>
    );
  }

  if (error) {
    return (
      <div className="center error-box">
        <h2>❌ Error</h2>
        <p>{error}</p>
        <button onClick={refreshData}>Try Again</button>
      </div>
    );
  }

  const enrolledCourses = courses.filter((course) =>
    enrollments.some((item) => item.courseId === course.id)
  );

  const completedCourses = enrollments.filter(
    (item) => item.status === "Completed"
  ).length;

  const inProgressCourses = enrollments.filter(
    (item) => item.status === "In Progress"
  ).length;

  return (
    <div className="app">
      <header>
        <div>
          <h1>Student Dashboard</h1>
          <p>Welcome back, {student.name}</p>
        </div>

        <div className="header-buttons">
          <button className="refresh" onClick={refreshData}>
            ↻ Refresh
          </button>

          <button className="logout" onClick={logout}>
            Logout
          </button>
        </div>
      </header>

      <main>
        <StudentCard />

        <section className="statistics">
          <div className="stat-card">
            <div className="stat-icon">📚</div>
            <div>
              <h2>{enrolledCourses.length}</h2>
              <p>Enrolled Courses</p>
            </div>
          </div>

          <div className="stat-card">
            <div className="stat-icon">✅</div>
            <div>
              <h2>{completedCourses}</h2>
              <p>Completed</p>
            </div>
          </div>

          <div className="stat-card">
            <div className="stat-icon">⏳</div>
            <div>
              <h2>{inProgressCourses}</h2>
              <p>In Progress</p>
            </div>
          </div>

          <div className="stat-card">
            <div className="stat-icon">🎓</div>
            <div>
              <h2>{courses.length}</h2>
              <p>Total Courses</p>
            </div>
          </div>
        </section>

        <section>
          <div className="section-title">
            <h2>My Courses</h2>
            <span>{enrolledCourses.length} Courses</span>
          </div>

          <div className="courses">
            {enrolledCourses.map((course) => {
              const enrollment = enrollments.find(
                (item) => item.courseId === course.id
              );

              return (
                <CourseCard
                  key={course.id}
                  course={course}
                  enrollment={enrollment}
                />
              );
            })}
          </div>
        </section>

        <section className="api-info">
          <h2>Mock API Status</h2>

          <div className="api-row">
            <span>Student API</span>
            <span className="online">● Connected</span>
          </div>

          <div className="api-row">
            <span>Course API</span>
            <span className="online">● Connected</span>
          </div>

          <div className="api-row">
            <span>Enrollment API</span>
            <span className="online">● Connected</span>
          </div>

          <div className="api-row">
            <span>State Management</span>
            <span className="online">● Context API</span>
          </div>
        </section>
      </main>

      <footer>
        <p>Full Stack Web Development | React + Mock API + Context API</p>
      </footer>
    </div>
  );
};

/* =========================
   MAIN APP
   ========================= */

const App = () => {
  const { isLoggedIn } = useContext(AppContext);

  return isLoggedIn ? <Dashboard /> : <Login />;
};

/* =========================
   RENDER APPLICATION
   ========================= */

  <AppProvider>
